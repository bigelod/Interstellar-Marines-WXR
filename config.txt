dominant_hand=right

deadzone=0.20
flipHands=false

# Shrinks the screen-anchored HUD (compass + squad info at the top, ammo / TP /
# health at the bottom) toward the centre of the view so nothing sits in the
# part of the frame a headset cannot show.
# Lower = smaller and further in. 0.60 was found on device on 2 Nov 2026.
# 1.0 disables the mechanism entirely. Re-read from disk every 2s, so you can
# tweak this while the game is running.
# World-space diamond waypoint markers are deliberately NOT affected.
hudScale=0.60

# View comfort. All live-reloaded every 2s, so tune them with the game running.
# eye_height / eye_forward move the VR camera off the helmet's own mount point.
# Raised from the original 0.08 because the Marine's own body was blocking the
# camera. hand_height lifts the weapon with the eye so the gun does not stay
# down by the chest where the body occludes it.
eye_height=0.16
eye_forward=0.05
hand_height=0.05

# Helmet lamp, relative to the VR camera. Above the middle of the view reads as
# helmet-mounted; below it reads as a torch held at the chest.
light_height=0.10
light_forward=0.10

# Near clip plane, metres. Low enough that a holographic sight can be brought
# right up to the eye without being cut off. Raise it toward 0.18 if distant
# geometry starts z-fighting.
near_clip=0.05

# Unity occlusion culling on the player camera. 1 is both the default and the fast
# path - culling is what skips the draw calls, so 0 costs frames rather than
# saving them.
# It was defaulted to 0 on 3 Nov because in this game occlusion culling does not
# merely skip draw calls, it switches the lights off: ToggleOcclusion.
# OnBecameInvisible disables every Light in a room that culls out, so when the
# camera ends up inside geometry the world is still drawn, just unlit. That was the
# pitch-black screen, and why it survived a respawn. The blackouts are gone as of
# the 3 Nov build and 1 behaves on device, so it is back on.
# 0 = lights stay on everywhere, at a real cost in a corridor game.
# 1 = the normal fast path, with the mod forcing it off per frame for as long as
#     the camera is detected inside geometry.
occlusion_culling=1

# View distance, metres. 80 originally, then 60; 45 because this is a horror game
# and nearly everything that matters is at corridor range.
# Not purely a draw-cost knob: ToggleOcclusion also unlights anything that
# frustum-culls out, so taking this too low makes distant rooms visibly light up
# as you walk toward them. Tune it on device from the VR Settings panel.
#
# THE FOG FOLLOWS THIS NUMBER. Linear fog runs from 45% to 95% of far_clip, so the
# view is walled off by murk just before the clip plane rather than having geometry
# pop through a hard edge. Change far_clip and the fog moves with it, live.
far_clip=45

# Damage feedback. In VR the game's full-screen red flash is suppressed entirely
# and replaced with two red bands in peripheral vision, the approach proven in
# Shogo-VR (commit ab012bc).
#
# The original had to go: MarineController draws it as one full-screen GUI.Box,
# and IMGUI runs once per CAMERA, not once per frame - with seven cameras in the
# scene a 0.25 tint composites to about 0.87, which fills a headset's whole field
# of view and reads as the lights going out rather than as a flash. The bands are
# drawn from the mod's own overlay camera, exactly once per frame.
#
# pain_alpha      peak opacity of the bands. 0 disables them.
# pain_bar_width  peak width of each band as a fraction of the VISIBLE width.
#                 Both width and opacity scale down as the hit fades.
# pain_bar_inset  how much of the panel WinlatorXR actually shows. The bands sit
#                 at the edge of that region and grow inward, so at the default
#                 Confirmed on device on 3 Nov that 1.00 is right: the panel is
#                 wider than it is tall, so the vertical crop is the binding one
#                 and the full frame width is in view.
pain_alpha=0.55
pain_bar_width=0.15
pain_bar_inset=1.00

# Scale applied to HelmetOccluder, the invisible shell around the head.
# 1.0 = untouched. This mesh is on layer 14 (UIOverlay), which only HelmetCamera
# draws, so it CANNOT affect the world view - it was never the black screen.
# Its shadow casting is now always off, which is the only way it reached the
# world. Only takes effect on the next spawn.
occluder_scale=1.0

# How wide an angle WinlatorXR paints the captured frame across, in degrees.
# This does NOT have to match the FOV the game renders with (98 vertical).
# Declaring LESS shows the SAME scene, shrunk toward the centre of the view, at
# full render resolution - no black borders, no lost pixels, and the sync pixel
# stays in the top-left corner. That makes it the cheap way to fit everything
# inside the part of the panel the headset actually shows.
# 90 is the value found on device on 2 Nov 2026 - confirmed working, and proof
# that the runtime does consume this field. 104.5 is the HWXR default.
#
# ADJUSTABLE IN THE HEADSET: open any menu or pause, click "VR Settings" in the
# upper-left of the content area, and use the -/+ buttons. Every setting in this
# file is in that panel, and every change is written straight back here.
screen_fov=90

# Render resolution, applied once at startup and again when changed from the VR
# Settings panel. Lower is a real win here - every pixel is rendered, captured,
# encoded and then repainted across the panel, so it is paid for more than once.
# 1280x960 is the owner's balance point. 1024x768 is the next step down.
# Set render_width=0 to leave the window alone entirely, which is the escape hatch
# if a resolution change ever upsets the HUD bend pipeline.
render_width=1280
render_height=960

# Supersampling for the visor HUD's own render target, written into
# HUDCamera.hudScale. That field is NOT a screen-size knob despite the name: the
# game sizes its two HUD render textures at Screen * hudScale AND divides the
# canvas scaler's 3840x2160 reference resolution by the same number, so raising it
# draws the same HUD, at the same apparent size, into proportionally more pixels.
# It is the one lever that sharpens the HUD text without changing its fit.
# 1.1 is the game's own default and the mod used to force 1.0, which is why the
# text read as low-res in the headset. 2.0 doubles the linear resolution.
# Cost: two render targets at 2x and two full-screen blits over 4x the pixels -
# wind this back first if the frame rate is short.
hud_supersample=2.0

# Shadows. Only the player's helmet lamp casts any at all - every other light in
# the level has its shadow casting switched off (VRLightTamer), because real-time
# shadows are the most expensive thing a light can do and the torch is the only one
# that tells you anything. These two govern that single light.
# shadow_resolution: 0=Low, 1=Medium, 2=High, 3=VeryHigh.
shadow_distance=15
shadow_resolution=0

# Edge length of the red sync patch in the top-left corner, in framebuffer pixels.
# XRAPI 0.4 asks for 10x10 and 10 is what it draws. It sat at 150 for a long while
# on the theory that a letterboxed panel might move the true corner; on device that
# bought nothing and was plainly visible to the player. Raise it only if the
# runtime ever stops reading the frame id.
sync_pixel_size=10
